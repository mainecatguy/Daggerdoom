// Defines the Doomerfall Controls submenu. Its entry in Options > Customize Controls is inserted
// next to the other submenus by Doomerfall_Events.UiTick (zscript/Doomerfall.zs), because
// AddOptionMenu can only append (below "Reset to defaults").
// All actions are unbound by default; bindings are saved in the player's config like any other.
// Interacting (Grab, Steal while crouching, talk, shops, containers) is Doom's own Use control, so it has no entry here.
// Damage numbers over hit monsters (zscript/DoomerfallDamageNumbers.zs). Off by default; on by default in a dev launch.
AddOptionMenu "HUDOptions"
{
	StaticText ""
	Option "Show damage numbers", "dd_damagenumbers", "OnOff"
}

OptionMenu "DoomerfallControlsMenu"
{
	Title "Doomerfall Controls"
	ScrollTop 2
	StaticTextSwitchable "$CNTRLMNU_SWITCHTEXT1", "$CNTRLMNU_SWITCHTEXT2", "ControlMessage"

	StaticText ""
	Control "Open Fast Travel", "netevent df_fast_travel"
	Control "Open Inventory", "netevent df_inventory"
	Control "Open Journal", "netevent df_journal"
	Control "Rest", "netevent df_rest"
	Control "Info", "netevent df_info"
	Control "Change Ammo", "netevent df_change_ammo"
}
